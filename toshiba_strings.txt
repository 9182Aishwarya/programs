Reduce the string by removing K consecutive identical characters

#include<stdio.h>
#include<string.h>
int main()
{
    char str[100];
    gets(str);
    int k;
    scanf("%d",&k);
    int i=0;
    while(str[i]!='\0')
    {
        int count=1;
        while(str[i+count]==str[i])
        count++;
        if(count>=k)
        {
            memmove(&str[i],&str[i+k],strlen(&str[i+k])+1);
            if(i>0)
            i--;
        }
        else
        i+=count;
    }
    printf("%s",str);
}

#include<stdio.h>
#include<string.h>
int main()
{
    int arr[100];
    int target;
    int n;
    scanf("%d",&n);
    for(int i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    scanf("%d",&target);
    for(int i=0;i<n-1;i++)
    {
        for(int j=i+1;j<n;j++)
        {
            if(arr[i]+arr[j]==target)
            {
                printf("%d %d\n",arr[i],arr[j]);
            }
        }
    }
}

#include <stdio.h>

int main()
{
    int a[100], b[100], c[200];
    int n1, n2;

    scanf("%d", &n1);
    for(int i = 0; i < n1; i++)
        scanf("%d", &a[i]);

    scanf("%d", &n2);
    for(int i = 0; i < n2; i++)
        scanf("%d", &b[i]);

    int i = 0, j = 0, k = 0;

    while(i < n1 && j < n2)
    {
        if(a[i] <= b[j])
            c[k++] = a[i++];
        else
            c[k++] = b[j++];
    }

    while(i < n1)
        c[k++] = a[i++];

    while(j < n2)
        c[k++] = b[j++];

    printf("Merged array:\n");

    for(i = 0; i < k; i++)
        printf("%d ", c[i]);

    return 0;
}

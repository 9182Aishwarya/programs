Check if the given String is Palindrome or not


#include<stdio.h>
#include<string.h>

int polindrome(char *str)
{
    int len=strlen(str)-1;
    int i=0;
    while(i<len)
    {
        if(str[i]!=str[len])
        return 0;
        i++;
        len--;
    }
    return 1;
}
int main()
{
    char str[100];
    fgets(str,100,stdin);
    str[strlen(str)-1]='\0';
    if(polindrome(str))
    printf("polindrome");
    else
    printf("not polindrome");
}


Count number of vowels, consonants, spaces in String

#include<stdio.h>
#include<string.h>
int main()
{

    for(int i=0;str[i];i++)
    {
        if(str[i]=='a'||str[i]=='e'||str[i]=='i'||str[i]=='o'||str[i]=='u'||str[i]=='A'||str[i]=='E'||str[i]=='I'||str[i]=='O'||str[i]=='U')

        }
        else if(str[i]==' ')
        whitespace++;
        else
        consonants++;
    }
    printf("%d %d %d",vowels,whitespace,consonants);
}

Longest Common Subsequence (LCS) using BITMASK

#include<stdio.h>
#include<string.h>
int issubsequence(char *sub,char *str)
{
    int i=0,j=0;
    while(sub[i]&&str[j])
    {
        if(sub[i]==str[j])
        i++;
        j++;
    }
    return sub[i]=='\0';
}
int main()
{
    char A[20],B[20];
    fgets(A,20,stdin);
    fgets(B,20,stdin);
    A[strlen(A)-1]='\0';
    B[strlen(B)-1]='\0';
    int n=strlen(A);
    char large[20];
    int max=0;
    for(int mask=1;mask<(1<<n);mask++)
    {
        int index=0;
        char sub[20];
        for(int i=0;i<n;i++)
        {
            if(mask&(1<<i))
            {

                sub[index++]=A[i];
            }
        }
        sub[index]='\0';
        if(issubsequence(sub,B))
        {
            if(index>max)
            {
                max=index;
                strcpy(large,sub);
            }
        }
    }
    printf("%s\n",large);
    printf("%d",max);
}

        scanf("%d",&arr[i]);
    }
    scanf("%d",&k);
    int count=0;
    for(int i=0;i<n;i++)
    {
        int sum=0;
        for(int j=i;j<n;j++)
        {
            sum+=arr[j];
            if(sum==k)
            {
                count++;
                for(int x=i;x<=j;x++)

                printf("\n");
            }
        }
    }
    printf("%d ",count);
}

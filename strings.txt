#include <stdio.h>
#include <string.h>

void reverse(char str[])
{
    int i = 0, j = strlen(str) - 1;
    char temp;

    while(i < j)
    {
        temp = str[i];
        str[i] = str[j];
        str[j] = temp;
        i++;
        j--;
    }
}

void myitoa(int n, char str[])
{
    int i = 0, sign = 0;

    if(n == 0)
    {
        str[i++] = '0';
        str[i] = '\0';
        return;
    }

    if(n < 0)
    {
        sign = 1;
        n = -n;
    }

    while(n > 0)
    {
        str[i++] = (n % 10) + '0';
        n /= 10;
    }

    if(sign)
        str[i++] = '-';

    str[i] = '\0';

    reverse(str);
}

int main()
{
    int n;
    char str[20];

    scanf("%d", &n);

    myitoa(n, str);

    printf("%s", str);

    return 0;
}

    scanf("%s", str);

    if(str[i] == '-')
    {
        sign = -1;
        i++;
    }

    // Integer part
    while(str[i] != '\0' && str[i] != '.')
    {
        if(str[i] < '0' || str[i] > '9')
        {
            printf("ERROR");
            return 0;
        }

        num = num * 10 + (str[i] - '0');
        i++;
    }

    // Decimal point must exist
    if(str[i] != '.')
    {
        printf("ERROR");
        return 0;
    }

    dot = 1;
    i++;

    // At least one digit after '.'
    if(str[i] == '\0')
    {
        printf("ERROR");
        return 0;
    }

    // Fraction part
    while(str[i] != '\0')
    {
        if(str[i] < '0' || str[i] > '9')
        {
            printf("ERROR");
            return 0;
        }

        num = num + (str[i] - '0') * frac;
        frac /= 10;
        i++;
    }

    printf("%f", sign * num);

    return 0;
}

Maximum sum of any contiguous subarray

Input: [2, 1, 5, 1, 3, 2], k=3 
Output: 9
Explanation: Subarray with maximum sum is [5, 1, 3].
Example 2:
Input: [2, 3, 4, 1, 5], k=2 
Output: 7
Explanation: Subarray with maximum sum is [3, 4].


#include <stdio.h>

int main()
{
    int n, a[100], k;
    int i, sum = 0, max;

    scanf("%d", &n);

    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    scanf("%d", &k);

    // Sum of first window
    for(i = 0; i < k; i++)
        sum += a[i];

    max = sum;

    // Slide the window
    for(i = k; i < n; i++)
    {
        sum = sum - a[i - k] + a[i];

        if(sum > max)
            max = sum;
    }

    printf("%d", max);

    return 0;
}


counting vowels nd consonants


#include <stdio.h>

int main()
{
    char str[100];
    int count[26] = {0};
    int i;

    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] >= 'a' && str[i] <= 'z')
            count[str[i] - 'a']++;

        else if(str[i] >= 'A' && str[i] <= 'Z')
            count[str[i] - 'A']++;
    }

    for(i = 0; i < 26; i++)
    {
        if(count[i] > 0)
            printf("%c is %d\n", 'a' + i, count[i]);
    }

    return 0;
}

Lexicographic rank of a String

#include <stdio.h>
#include <string.h>

int fact(int n)
{
    int f = 1;
    for (int i = 2; i <= n; i++)
        f *= i;
    return f;
}

int lexicographicRank(char str[])
{
    int n = strlen(str);
    int rank = 1;

    for (int i = 0; i < n; i++)
    {
        int count = 0;

        // Count characters smaller than str[i] to its right
        for (int j = i + 1; j < n; j++)
        {
            if (str[j] < str[i])
                count++;
        }

        rank += count * fact(n - i - 1);
    }

    return rank;
}

int main()
{
    char str[100];

    printf("Enter string: ");
    scanf("%s", str);

    printf("Rank = %d\n", lexicographicRank(str));

    return 0;
}
